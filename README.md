🚀 Setup e Execução
cd backend
# Criar ambiente virtual
python -m venv venv

# Ativar ambiente (Windows)
venv\Scripts\activate

# Instalar dependências
pip install -r requirements.txt

# Executar servidor
uvicorn main:app --reload

Frontend (Node.js/React)
cd frontend
# Instalar dependências
npm install

# Executar servidor de desenvolvimento
npm run dev
